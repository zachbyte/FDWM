# Fedora packages FDWM needs; install.sh and update.sh both read this list.

# building dwm, st, dmenu and slock
git gcc make pkgconf-pkg-config diffutils tar xz
libX11-devel libXft-devel libXrender-devel
fontconfig-devel freetype-devel
libXext-devel libXrandr-devel libxcrypt-devel
# dwm's window swallowing (xcb-res)
libxcb-devel

# X and the session started by .xinitrc
xorg-x11-server-Xorg xorg-x11-xinit xorg-x11-drv-libinput
xsetroot gnome-keyring mate-polkit dejavu-sans-mono-fonts
xset xss-lock
# awk, which fdwm-bar and fdwm-theme use (not every Fedora install has it)
gawk
# pgrep and pkill (procps-ng), which fdwm-bar, fdwm-menu and fdwm-theme use
# (Fedora's container image doesn't have them)
procps-ng

# notifications: dunst shows them, notify-send (libnotify) sends them
dunst libnotify

# sound, and the volume, brightness and media keys; pactl (pulseaudio-utils)
# tells fdwm-bar when the volume changes
pipewire wireplumber pipewire-pulseaudio pulseaudio-utils brightnessctl playerctl
# udevadm (systemd-udev), which tells fdwm-bar when a charger is plugged in
# or out
systemd-udev

# battery life: TLP sets the CPU, Wi-Fi, USB, PCIe and disks to save power
# on battery and back on the charger (install.sh removes tuned, tuned-ppd and
# power-profiles-daemon first, which would fight it)
tlp

# screenshots (fdwm-shot): maim takes them, xclip copies them
maim xclip

# nnn, the terminal file manager
nnn

# Thunar, the graphical file manager, with gvfs for its trash and drives
Thunar gvfs

# Zen Browser, which install.sh installs from Flathub with flatpak (Fedora
# doesn't package it)
flatpak

# dark mode: GTK apps (Thunar) read it from ~/.config/gtk-3.0 and gtk-4.0;
# Flatpak apps (Zen) ask the settings portal, which xdg-desktop-portal-gtk
# answers from gsettings' color-scheme, kept in dconf
xdg-desktop-portal xdg-desktop-portal-gtk gsettings-desktop-schemas dconf glib2

# neovim, plus what its plugins and language servers need
# (/usr/bin/npm is named by path because its package name differs between releases)
neovim ripgrep unzip tree-sitter-cli /usr/bin/npm java-latest-openjdk-headless
